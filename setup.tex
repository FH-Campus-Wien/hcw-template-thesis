% Zentrale Auswahl der Thesis-Variante und des Textsatzes.
% Erlaubte Werte:
%   Sprache:       de oder en
%   Abschlussart:  ba oder ma
%   Textausrichtung: blocksatz oder flattersatz

\newcommand{\HCWLanguage}{en}
\newcommand{\HCWThesisLevel}{ma}
\newcommand{\HCWTextAlignment}{flattersatz}

% Die offiziellen Referenzvorlagen verwenden diese akademischen Grade.
% Falls ein Studiengang einen anderen Grad verleiht, wird nur die betreffende
% Zeile angepasst; die Auswahl zwischen BA und MA erfolgt weiterhin oben.
\newcommand{\HCWBachelorDegree}{Bachelor of Science in Engineering}
\newcommand{\HCWMasterDegree}{Master of Science in Engineering}

% Ab hier sind keine Anpassungen erforderlich.
\RequirePackage{etoolbox}

\newif\ifHCWEnglish
\newif\ifHCWMaster
\newif\ifHCWJustified

\ifdefstring{\HCWLanguage}{de}{%
  \HCWEnglishfalse
}{%
  \ifdefstring{\HCWLanguage}{en}{%
    \HCWEnglishtrue
  }{%
    \PackageError{hcw-thesis}{Unknown language '\HCWLanguage'}%
      {Use de or en in setup.tex.}%
  }%
}

\ifdefstring{\HCWThesisLevel}{ba}{%
  \HCWMasterfalse
}{%
  \ifdefstring{\HCWThesisLevel}{ma}{%
    \HCWMastertrue
  }{%
    \PackageError{hcw-thesis}{Unknown thesis level '\HCWThesisLevel'}%
      {Use ba or ma in setup.tex.}%
  }%
}

\ifdefstring{\HCWTextAlignment}{blocksatz}{%
  \HCWJustifiedtrue
}{%
  \ifdefstring{\HCWTextAlignment}{flattersatz}{%
    \HCWJustifiedfalse
  }{%
    \PackageError{hcw-thesis}{Unknown text alignment '\HCWTextAlignment'}%
      {Use blocksatz or flattersatz in setup.tex.}%
  }%
}

  \newcommand{\HCWSupervisorsLabel}{betreut von:}
  \newcommand{\HCWSubmissionPlace}{Wien}
  \ifHCWMaster
    \newcommand{\HCWThesisName}{Masterarbeit}
  \else
    \newcommand{\HCWThesisName}{Bachelorarbeit}
  \fi
