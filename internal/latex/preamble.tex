% Die Satzspiegel der vier V1-Word-Vorlagen sind abschnittsabhängig.
% Kopf- und Fußzeilen liegen außerhalb der hier definierten Textränder.
\newcommand{\HCWTextArea}[2]{%
  \newgeometry{%
    top=25mm,bottom=#1,left=25mm,right=25mm,%
    includehead=false,includefoot=false,%
    headheight=18pt,headsep=\dimexpr 12.5mm-18pt\relax,%
    footskip=#2%
  }%
}

\newcommand{\HCWMainMatterLayout}{\HCWTextArea{25mm}{12.5mm}}
\newcommand{\HCWBackMatterLayout}{\HCWTextArea{20mm}{7.5mm}}

  \setlength{\parindent}{0pt}%
  \setlength{\parskip}{8pt}%
  \clubpenalty=10000%
  \widowpenalty=10000%
  \displaywidowpenalty=10000%
}
